async function fetchFromAdzuna() { const domainInput = document.getElementById('domain').value; const locationInput = document.getElementById('location').value; const loading = document.getElementById('loading'); const loadMoreBtn = document.getElementById('loadMore'); loading.style.display = 'block'; loadMoreBtn.style.display = 'none'; // On simplifie la recherche pour Adzuna (on enlève les termes trop restrictifs) const query = encodeURIComponent(domainInput); const place = encodeURIComponent(locationInput); // URL vers Adzuna France const url = `https://api.adzuna.com/v1/api/jobs/fr/search/${currentPage}?app_id=${APP_ID}&app_key=${APP_KEY}&results_per_page=15&what=${query}&where=${place}&content-type=application/json`; try { const response = await fetch(url); const data = await response.json(); if (data.results && data.results.length > 0) { data.results.forEach(job => { // On vérifie si c'est une alternance dans la description pour mettre une étoile const isAlternance = job.description.toLowerCase().includes('alternance') || job.title.toLowerCase().includes('alternance'); const div = document.createElement('div'); div.className = 'job-card'; div.innerHTML = `

${job.title} ${isAlternance ? '🌟' : ''}

🏢 ${job.company.display_name} | 📍 ${job.location.display_name}

${job.description.substring(0, 150)}...

Voir l'offre → `; document.getElementById('results').appendChild(div); allJobs.push(job); }); loadMoreBtn.style.display = 'inline-block'; } else { if(currentPage === 1) { document.getElementById('results').innerHTML = `

😕 Aucun résultat précis pour "${domainInput}".
Essaie avec un mot plus simple comme "Vente" ou "Comptabilité".

`; } } } catch (error) { alert("Erreur : Vérifie que ton ID et ta Clé Adzuna sont bien collés sans espaces !"); console.error(error); } finally { loading.style.display = 'none'; // Mise à jour des stats document.getElementById('totalJobs').textContent = allJobs.length; } }
